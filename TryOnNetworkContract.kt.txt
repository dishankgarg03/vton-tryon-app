package com.example.vtonapp

import okhttp3.MultipartBody
import okhttp3.RequestBody
import retrofit2.Response
import retrofit2.http.Multipart
import retrofit2.http.POST
import retrofit2.http.Part

data class TryOnResponse(
    val success: Boolean,
    val resultImageUrl: String?,
    val error: String?
)

interface TryOnApiService {
    @Multipart
    @POST("api/tryon")
    suspend fun uploadTryOnTask(
        @Part personImage: MultipartBody.Part,
        @Part garmentImage: MultipartBody.Part,
        @Part("category") category: RequestBody
    ): Response<TryOnResponse>
}
